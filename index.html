export function longestStreak(days: Date[]) {
  const seen = new Set(days.map(d => key(d)));
  let best = 0;
  for (const d of seen) {
    if (seen.has(prev(d))) continue;
    let run = 1, cur = d;
    while (seen.has(cur = next(cur))) run++;
    best = Math.max(best, run);
  }
  return best; // O(n) ✨
}
